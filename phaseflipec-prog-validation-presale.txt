class CSVLogger:
    def __init__(self, filename):
        self.filename = filename
        self.file = open(filename, 'w', newline='')
        self.writer = csv.writer(self.file)
        self.writer.writerow([
            'Timestamp', 'Test_Name', 'Status', 'Job_ID', 
            'Qubits', 'Distance', 'Entropy', 'Top_State_Prob', 
            'Unique_States', 'Shots', 'Quality_Score', 'Message'
        ])
        self.file.flush()
    
    def log(self, test_name, status, job_id='', qubits=0, distance=0, 
            entropy=0.0, top_prob=0.0, unique_states=0, shots=0, 
            quality=0.0, message=''):
        self.writer.writerow([
            datetime.now().isoformat(),
            test_name,
            status,
            job_id,
            qubits,
            distance,
            f"{entropy:.4f}",
            f"{top_prob:.4f}",
            unique_states,
            shots,
            f"{quality:.4f}",
            message
        ])
        self.file.flush()
    
    def close(self):
        self.file.close()

class TestResults:
    def __init__(self, csv_logger):
        self.passed = 0
        self.failed = 0
        self.warnings = 0
        self.tests = []
        self.start_time = time.time()
        self.csv = csv_logger
    
    def record(self, name: str, passed: bool, message: str = "", 
               job_id: str = "", entropy: float = 0.0, top_prob: float = 0.0,
               quality: float = 0.0, qubits: int = 0, distance: int = 0,
               unique_states: int = 0):
        status = "PASS" if passed else "FAIL"
        timestamp = datetime.now().strftime("%H:%M:%S")
        
        self.tests.append({
            'name': name,
            'status': status,
            'passed': passed,
            'message': message
        })
        
        # Log to CSV in real-time
        self.csv.log(
            test_name=name,
            status=status,
            job_id=job_id,
            qubits=qubits,
            distance=distance,
            entropy=entropy,
            top_prob=top_prob,
            unique_states=unique_states,
            shots=SHOTS,
            quality=quality,
            message=message
        )
        
        icon = "✅" if passed else "❌"
        print(f"  [{timestamp}] {icon} {name}")
        if message:
            print(f"           → {message}")
        
        if passed:
            self.passed += 1
        else:
            self.failed += 1
    
    def summary(self):
        elapsed = time.time() - self.start_time
        total = self.passed + self.failed
        pass_rate = (self.passed / total * 100) if total > 0 else 0
        
        print("\n" + "="*70)
        print("📊 PHASEFLIP QEC PRE-SALE TEST SUMMARY")
        print("="*70)
        print(f"Total Tests:    {total}")
        print(f"Passed:         {self.passed} ({pass_rate:.1f}%)")
        print(f"Failed:         {self.failed}")
        print(f"Duration:       {elapsed:.2f}s")
        print(f"Results saved:  {CSV_FILE}")
        print("="*70)
        
        return pass_rate >= 70

def run_and_retrieve(qc, test_name, csv_logger):
    """Submit job and wait for results with real-time logging"""
    print(f"   Optimizing...", end='', flush=True)
    qc_opt = qiskit_transpile(qc, basis_gates=['rz', 'sx', 'x', 'cz', 'measure'], 
                               optimization_level=1)
    
    print(" Converting...", end='', flush=True)
    try:
        converted_program = qbraid_transpile(qc_opt, target='braket')
    except Exception:
        converted_program = qbraid_transpile(qc_opt, target='openqasm3')
    
    print(" Submitting...", end='', flush=True)
    job = backend.run(converted_program, shots=SHOTS)
    job_id = job.id
    print(f" Job ID: {job_id[:8]}...")
    
    # Log job submission
    csv_logger.log(
        test_name=test_name,
        status="SUBMITTED",
        job_id=job_id,
        qubits=qc.num_qubits,
        shots=SHOTS,
        message="Job submitted to IQM Garnet"
    )
    
    # Wait for completion
    print("   Waiting for results...", end='', flush=True)
    status = job.status()
    wait_count = 0
    while status.name not in ['COMPLETED', 'FAILED', 'CANCELLED']:
        time.sleep(5)
        wait_count += 1
        if wait_count % 6 == 0:  # Update every 30s
            print(f"\n   Still waiting ({wait_count*5}s)...", end='', flush=True)
        status = job.status()
    
    if status.name == 'COMPLETED':
        res = job.result()
        counts = res.data.get_counts() if hasattr(res.data, 'get_counts') else res.get_counts()
        print(" ✓ Retrieved")
        
        # Log completion
        csv_logger.log(
            test_name=test_name,
            status="COMPLETED",
            job_id=job_id,
            qubits=qc.num_qubits,
            shots=SHOTS,
            unique_states=len(counts),
            message=f"Results received: {len(counts)} unique states"
        )
        
        return counts, job_id
    else:
        print(f" ✗ {status.name}")
        csv_logger.log(
            test_name=test_name,
            status="FAILED",
            job_id=job_id,
            message=f"Job failed with status: {status.name}"
        )
        return {}, job_id

def calculate_entropy(counts):
    """Shannon entropy of measurement results"""
    total = sum(counts.values())
    if total == 0:
        return 0
    probs = np.array([c/total for c in counts.values()])
    probs = probs[probs > 0]
    return -np.sum(probs * np.log2(probs))

def get_top_state_probability(counts):
    """Get probability of most common state"""
    if not counts:
        return 0
    total = sum(counts.values())
    max_count = max(counts.values())
    return max_count / total

def calculate_quality_score(counts, num_qubits):
    """
    Quality score for QEC circuit performance
    Higher entropy = better (shows entanglement)
    More unique states = better (shows quantum behavior)
    """
    if not counts:
        return 0
    
    entropy = calculate_entropy(counts)
    max_entropy = num_qubits  # Maximum possible entropy
    unique_ratio = len(counts) / (2**num_qubits)  # Fraction of possible states observed
    
    # Quality = weighted combination
    quality = (entropy / max_entropy) * 0.7 + unique_ratio * 0.3
    return quality
